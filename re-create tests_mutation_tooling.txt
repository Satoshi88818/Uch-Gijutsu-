# ------------------------------------------------------------------- lesions
def test_lesions_zero_exactly_what_they_say():
    sim, _ = rand_write_phase(CFG_F, 0)
    assert np.abs(sim.u).max() > 0 and np.abs(sim.s).max() > 0 and np.abs(sim.H).max() > 0
    h, t = sim.H.copy(), sim.t
    fast = sim.clone(); apply_lesion(fast, "fast")
    assert not fast.u.any() and not fast.u_prev.any() and not fast._f.any()                  # all time levels
    assert np.abs(fast.s).max() > 0                                                            # the slow field survives
    np.testing.assert_array_equal(fast.H, h); assert fast.t == t
    both = sim.clone(); apply_lesion(both, "both")
    assert not both.u.any() and not both.s.any() and not both._s.any()
    np.testing.assert_array_equal(both.H, h)
    assert np.abs(sim.u).max() > 0                                                             # the original is untouched
    with pytest.raises(ValueError):
        apply_lesion(sim, "everything")
    zero_fast_field(sim); assert not sim._f.any()

@pytest.fixture(scope="module")

def test_hologram_state_decays_exactly_and_is_independent_of_the_fields(snaps):
    out, clips = snaps
    decay = CFG_F.base.medium.holo_decay
    H = out["HOLO"]["H"]
    np.testing.assert_allclose(H[1], H[0] * decay ** 100, rtol=1e-4, atol=1e-9)
    np.testing.assert_allclose(H[2], H[0] * decay ** 200, rtol=1e-4, atol=1e-9)
    assert np.abs(H[0]).max() > 0.01 and clips == 0
    for d in range(3):
        np.testing.assert_array_equal(out["HOLO_SLOW"]["H"][d], H[d])                         # same hologram, different fields


def test_each_lesion_leaves_only_its_own_state(snaps):
    out, _ = snaps
    assert not out["HOLO"]["s"].any()                                                          # fields zeroed: nothing drives s
    assert not out["SLOW"]["H"].any()                                                          # hologram off: H is exactly 0
    for lesion in ("HOLO_SLOW", "SLOW"):
        assert np.abs(out[lesion]["s"][0]).max() > 0 and not np.allclose(out[lesion]["s"][0], out[lesion]["s"][2])
    assert not np.allclose(out["SLOW"]["s"][0], out["HOLO_SLOW"]["s"][0])                      # the hologram changes the write dynamics
    assert set(out) == set(LESIONS) and all(v["H"].shape == (3, 361) for v in out.values())


def test_incremental_delay_stepping_equals_a_single_run_and_is_deterministic(snaps):
    out, _ = snaps
    direct, _ = trial_snapshots(CFG_F, CFG_R, 0, delays=(0, 200))
    for lesion in LESIONS:
        np.testing.assert_array_equal(direct[lesion]["s"][1], out[lesion]["s"][2])
        np.testing.assert_array_equal(direct[lesion]["H"][1], out[lesion]["H"][2])
    again, _ = trial_snapshots(CFG_F, CFG_R, 0, delays=(0, 100, 200))
    np.testing.assert_array_equal(again["SLOW"]["s"], out["SLOW"]["s"])


def test_hologram_state_at_250_steps_matches_the_v11_state_capture():
    out, _ = trial_snapshots(CFG_F, CFG_R, 0, delays=(0, 250))
    v11, _ = snapshot_trial(CFG_F, 0)

    assert not np.allclose(out["HOLO_SLOW"]["s"][1], v11["s"])                                  # zeroing the fast field changes s

def test_cache_roundtrip_stale_rejection_and_key_sensitivity(tmp_path, monkeypatch):
    calls = []

    def fake(cfg_full, cfg_ring, delays=DELAYS, verbose=False):
        calls.append(tuple(delays))
        n, D = cfg_full.n_trials, len(delays)
        rng = np.random.default_rng(len(calls))
        arr = {"clip_events": 0.0, "delays": np.array(delays)}
        for name in LESIONS:
            arr[f"H_{name}"] = rng.normal(size=(n, D, 361)); arr[f"s_{name}"] = rng.normal(size=(n, D, 361))
        return arr

    monkeypatch.setattr(rt, "build_retention_arrays", fake)
    a1, c1 = get_retention_arrays(0, 24, (0, 100), tmp_path)
    a2, c2 = get_retention_arrays(0, 24, (0, 100), tmp_path)
    assert (c1, c2) == (False, True) and calls == [(0, 100)]
    np.testing.assert_array_equal(a1["H_HOLO"], a2["H_HOLO"]); assert a2["clip_events"] == 0.0
    _, c3 = get_retention_arrays(0, 24, (0, 200), tmp_path)                                   # other delays -> stale -> rebuilt
    assert c3 is False and len(calls) == 2
    assert retention_key(CFG_F, (0, 100)) != retention_key(CFG_F, (0, 200)) != retention_key(rand_arm_config("FULL", 4, 24), (0, 200))


def test_mod_features_use_the_medium_coefficients():
    rng = np.random.default_rng(0)
    arr = {"H_HOLO": rng.normal(size=(5, 2, 361)), "s_HOLO": rng.normal(size=(5, 2, 361))}
    med = CFG_F.base.medium
    np.testing.assert_allclose(mod_features(arr, "HOLO", 1, med), 0.3 * arr["s_HOLO"][:, 1] + 0.5 * arr["H_HOLO"][:, 1])
    m2 = replace(med, slow_mod=0.1, holo_mod=0.2)
    np.testing.assert_allclose(mod_features(arr, "HOLO", 0, m2), 0.1 * arr["s_HOLO"][:, 0] + 0.2 * arr["H_HOLO"][:, 0])


# ------------------------------------------------------- half-life and verdict
D5 = (0, 250, 500, 1000, 2000)


def _cell(r2, lo=0.05):
    return {"r2": r2, "r2_lo": r2 - 0.05, "r2_hi": r2 + 0.05, "diff_vs_null": {"diff": r2, "lo": lo, "hi": r2 + 0.1}}


def _sres(h=(0.2, 0.2, 0.18, 0.15, 0.12), s=(0.5, 0.3, 0.15, 0.05, 0.0), lo=0.05, hs_lo=0.05, null=-0.01, pc=0.999,
          mp=0.4, clip=0.0, drop=()):
    cells = {}
    for name, vals in (("HOLO", h), ("SLOW", s), ("HOLO_SLOW", s)):
        cells[name] = {str(d): {m: {r: _cell(v, lo) for r in ("linear", "rbf")} for m in ("1.0", "4.0")} for d, v in zip(D5, vals)}
    for d in drop:
        cells.pop(d)
    return {"cells": cells, "delays": list(D5), "clip_events": clip,
            "null": {m: {r: {"r2": null} for r in ("linear", "rbf")} for m in ("1.0", "4.0")},
            "hs": {str(d): {m: {r: {"point": 0.0, "lo": hs_lo, "hi": 0.2} for r in ("linear", "rbf")} for m in ("1.0", "4.0")} for d in D5},
            "baselines": {"fft_concat": {"shape": {"linear": {"r2": pc}}}, "map": {"position": {"linear": {"r2": mp}}}}}


def test_half_life_interpolates_and_flags_no_decay_or_no_signal():
    s = _sres(s=(0.4, 0.3, 0.15, 0.05, 0.0))
    assert half_life(s, "SLOW") == pytest.approx(250 + (0.3 - 0.2) / (0.3 - 0.15) * 250)       # between 250 and 500
    assert half_life(_sres(h=(0.2, 0.2, 0.19, 0.18, 0.15)), "HOLO") == ">2000"                  # never halves
    assert half_life(_sres(s=(0.05, 0.04, 0.03, 0.02, 0.0)), "SLOW") is None                    # nothing to halve


def test_best_r2_counts_a_cell_only_if_its_interval_versus_null_is_above_zero():
    assert best_r2(_sres(), "HOLO", 250, "1.0") == (0.2, "linear")
    assert best_r2(_sres(lo=-0.01), "HOLO", 250, "1.0")[0] == 0.0


def test_verdict_invalid_inconclusive_and_controls():
    assert verdict(_sres(clip=2.0))["outcome"] == "INVALID_CLIPPED"
    assert verdict(_sres(drop=("SLOW",)))["outcome"] == "INCONCLUSIVE"
    assert verdict(_sres(pc=0.4))["outcome"] == "POSITIVE_CONTROL_FAILS"
    leaky = _sres(); leaky["null"]["4.0"]["rbf"]["r2"] = 0.2                                   # a leak at ANY noise level counts
    assert verdict(leaky)["outcome"] == "CONTROL_FAILS"
    assert verdict(_sres(null=0.04))["outcome"] != "CONTROL_FAILS"
    assert verdict(_sres(mp=0.1))["outcome"] == "STATE_CONTROL_FAILS"


def test_verdict_hologram_floor_extends_memory_slow_persists_and_both_fade():
    assert verdict(_sres(h=(0.08, 0.07, 0.05, 0.04, 0.02)))["outcome"] == "HOLO_NEVER_ABOVE_FLOOR"
    ext = verdict(_sres())
    assert ext["outcome"] == "PLASTICITY_EXTENDS_MEMORY" and set(ext["half_lives"]) == set(DECODED)
    assert verdict(_sres(hs_lo=-0.01))["outcome"] != "PLASTICITY_EXTENDS_MEMORY"              # not distinguishable from the slow field
    assert verdict(_sres(hs_lo=-0.01, s=(0.5, 0.4, 0.3, 0.2, 0.15)))["outcome"] == "SLOW_PERSISTS"
    assert verdict(_sres(h=(0.2, 0.15, 0.1, 0.099, 0.05), s=(0.5, 0.2, 0.05, 0.02, 0.0)))["outcome"] == "BOTH_FADE"
    assert verdict(_sres(h=(0.2, 0.2, 0.2, 0.2, 0.101)))["outcome"] == "PLASTICITY_EXTENDS_MEMORY"   # threshold at the longest delay
    assert verdict(_sres(h=(0.2, 0.2, 0.2, 0.2, 0.099), s=(0.5, 0.3, 0.2, 0.2, 0.0)))["outcome"] == "BOTH_FADE"


def test_verdict_counts_the_hologram_only_with_a_positive_interval_and_can_use_another_noise_level():
    assert verdict(_sres(lo=-0.01))["outcome"] == "HOLO_NEVER_ABOVE_FLOOR"
    s = _sres()
    for d in D5:
        s["cells"]["HOLO"][str(d)]["4.0"] = {r: _cell(0.02) for r in ("linear", "rbf")}
    assert verdict(s, "1.0")["outcome"] == "PLASTICITY_EXTENDS_MEMORY" and verdict(s, "4.0")["outcome"] == "HOLO_NEVER_ABOVE_FLOOR"

def test_comparison_levels(tiny, monkeypatch):
    res, kw, cache = tiny
    seen = []
    real = rt.bootstrap_r2

    def spy(E_a, Y, E_b=None, n_boot=2000, seed=0, level=0.95):
        seen.append((E_b is not None, round(level, 6)))
        return real(E_a, Y, E_b, n_boot, seed, level)

    monkeypatch.setattr(rt, "bootstrap_r2", spy)
    run_retention(**kw)
    paired = {lv for p, lv in seen if p}
    assert paired == {round(1 - 0.05 / 6, 6), round(1 - 0.05 / 2, 6)}                           # vs NULL (3 lesions x 2 readouts), vs SLOW (2 readouts)
